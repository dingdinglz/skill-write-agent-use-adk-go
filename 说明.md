## 使用的模型应该是可重试的

你需要包裹一个retryLLM，把retryLLM传给agent，参考如下：

``` go
retryLLM := &RetryLLM{
    llm:        llmModel,
    maxRetries: 3,
}

type RetryLLM struct {
	llm        adkmodel.LLM
	maxRetries int
}

// Name 返回LLM名称
func (r *RetryLLM) Name() string {
	return r.llm.Name()
}

// GenerateContent 生成内容,遇到错误自动重试
func (r *RetryLLM) GenerateContent(ctx context.Context, req *adkmodel.LLMRequest, stream bool) iter.Seq2[*adkmodel.LLMResponse, error] {
	return func(yield func(*adkmodel.LLMResponse, error) bool) {
		// 尝试调用原始LLM
		for attempt := 0; attempt <= r.maxRetries; attempt++ {
			if attempt > 0 {
				// 重试前等待(指数退避)
				waitTime := (1 << uint(attempt-1)) * 2 // 2秒, 4秒, 8秒...
				time.Sleep(time.Duration(waitTime) * time.Second)
			}

			// 调用底层LLM
			hasError := false
			var lastError error

			for resp, err := range r.llm.GenerateContent(ctx, req, stream) {
				if err != nil {
					// 记录错误
					lastError = err
					hasError = true

					if attempt < r.maxRetries {
						// 还有重试机会,记录错误但不传递给调用者
						break // 跳出当前迭代,进行重试
					} else {
						// 已经是最后一次尝试,传递错误
						if !yield(resp, err) {
							return
						}
					}
				} else {
					// 成功的响应,传递给调用者
					if attempt > 0 {
					}
					if !yield(resp, nil) {
						return
					}
				}
			}

			// 如果没有错误,成功完成
			if !hasError {
				return
			}

			// 如果已达到最大重试次数,结束
			if attempt >= r.maxRetries {
				return
			}
		}
	}
}
```

## 创建一个agent

参考：

``` go
import (
	"google.golang.org/adk/agent/llmagent"
	"google.golang.org/adk/tool"
	"google.golang.org/genai"
)

NewAgent, err := llmagent.New(llmagent.Config{
    Name:        constants.ADKAppName,
    Model:       retryLLM, // 使用带重试机制的模型
    Description: "", // agent的描述
    Instruction: "", // agent的系统提示词
    Tools:      []tool.Tool{} // tools
    GenerateContentConfig: &genai.GenerateContentConfig{
        MaxOutputTokens: maxTokens, // 使用模型配置的 max_tokens
    },
})
```

## 运行agent

运行agent，你需要使用runner，参考

先创建runner：

``` go
import (
	adkagent "google.golang.org/adk/agent"
	"google.golang.org/adk/runner"
)

runConfig := adkagent.RunConfig{
    StreamingMode: adkagent.StreamingModeSSE,
}

newRunner, err := runner.New(runner.Config{
    AppName:        constants.ADKAppName,
    Agent:          NewAgent,
    SessionService: SessionService,
})
```

SessionService由session.InMemoryService()创建（"google.golang.org/adk/session"），可以根据session_id保存对话历史

然后运行：

``` go
// 处理流式响应
fullResponse:= ""

// userID和sessionID确定对话，可以寻找到历史聊天记录，userContent就是传入的message
for event, err := range newRunner.Run(context, userID, sessionID, userContent, runConfig) {
    if err != nil {
        break
    }

    if event == nil {
        continue
    }

    // 提取文本
    if event.Content != nil {
        for _, part := range event.Content.Parts {
            if part.Text != "" && event.Partial {
                content := part.Text
                fullResponse += content
            }
        }
    }
}
```

注意，为了内存考虑，如果后续不需要进行多轮对话，你需要在session.InMemoryService()删除对话，参考：

``` go
func cleanupSession(userID, sessionID string) error {
	sessionSvc := l.svcCtx.SessionService
	if sessionSvc == nil {
		return nil // Session service 不可用，跳过清理
	}

	// 删除 session
	err := sessionSvc.Delete(l.ctx, &session.DeleteRequest{
		AppName:   constants.ADKAppName,
		UserID:    userID,
		SessionID: sessionID,
	})
	if err != nil {
		return fmt.Errorf("failed to delete session: %w", err)
	}

	return nil
}
```
